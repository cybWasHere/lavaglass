cmake_minimum_required(VERSION 3.20)
project(glass-dock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick DBus)
find_package(KF6WindowSystem REQUIRED)
add_executable(glass-dock main.cpp)
target_compile_definitions(glass-dock PRIVATE DOCK_DIR="${CMAKE_SOURCE_DIR}")
target_link_libraries(glass-dock PRIVATE Qt6::Quick Qt6::DBus KF6::WindowSystem)
